def score_call_connect(event: dict, glance_data: dict) -> int:
    score = 0
    
    # 1. Fan-out (Only penalize extreme, bot-like behavior, not normal business)
    calls_today = event.get("numbersTheCallerHasRungToday", 0)
    if calls_today > 25:
        score += 20  # High volume, warrants attention
    elif calls_today > 10:
        score += 5   # Moderate volume, minor weight

    # 2. SIM Age (New SIMs are risky, but not guilty on their own)
    caller_sim_registered_at = event.get("callerSimRegisteredAt")
    if is_registered_today(caller_sim_registered_at):
        score += 15

    # 3. Relationship (Cold calls are normal, but add slight baseline risk)
    if event.get("callsBetweenThemBefore", 0) == 0:
        score += 5

    # 4. Glance Threat Intel (The strongest signal)
    if glance_data.get("risk_level") == "HIGH":
        score += 40
    elif glance_data.get("reports", 0) >= 2:
        score += 25

    return min(100, score)

def get_call_decision(score: int) -> dict:
    if score >= 75:

        return {"decision": "allow"}
